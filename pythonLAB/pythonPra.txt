Q2Develop a python module with functions to perform  Mean, Median and Mode operation.
# stats_utils.py

def mean(numbers):
    if len(numbers) == 0:
        return None
    return sum(numbers) / len(numbers)


def median(numbers):
    if len(numbers) == 0:
        return None

    nums = sorted(numbers)
    n = len(nums)
    mid = n // 2

    if n % 2 == 0:
        return (nums[mid - 1] + nums[mid]) / 2
    else:
        return nums[mid]


def mode(numbers):
    if len(numbers) == 0:
        return None

    freq = {}
    for num in numbers:
        freq[num] = freq.get(num, 0) + 1

    max_freq = max(freq.values())
    modes = [num for num, f in freq.items() if f == max_freq]

    if len(modes) == len(freq):
        return None   # no mode (all values unique)

    return modes[0] if len(modes) == 1 else modes

Q4. Develop a Python program to create a database name student. With the collection name studentinfo. Add 5 douments in the collection. Perform  read, update, and delete operations on documents in a MongoDB.

from pymongo import MongoClient

# connect to MongoDB
client = MongoClient("mongodb://localhost:27017/")
db = client.student    # database
collection = db.studentinfo   # collection

# ----- Add 5 documents -----
students = [
    {"name": "Amit", "age": 20},
    {"name": "Riya", "age": 21},
    {"name": "Sam", "age": 19},
    {"name": "Neha", "age": 22},
    {"name": "John", "age": 20}
]

collection.insert_many(students)
print("5 documents inserted.")

# ----- Read -----
print("All documents:")
for s in collection.find():
    print(s)

# ----- Update -----
collection.update_one({"name": "Amit"}, {"$set": {"age": 25}})
print("Amit updated.")

# ----- Delete -----
collection.delete_one({"name": "John"})
print("John deleted.")

Q3. Write a program to create a class Shape with methods called get_perimeter and get_area. Create a subclass called Circle that overrides the get_perimeter and get_area methods to calculate the area and perimeter of a circle.

class Shape:
    def get_perimeter(self):
        return 0

    def get_area(self):
        return 0


class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius

    def get_perimeter(self):
        return 2 * 3.14 * self.radius

    def get_area(self):
        return 3.14 * self.radius * self.radius


# Example
c = Circle(5)
print("Perimeter:", c.get_perimeter())
print("Area:", c.get_area())
